## Top Nix DSL idioms

Common patterns used across Kix packages and cluster definitions:

```nix
# Reference a resource name while tracking the dependency:
self.service.out.name

# Reference a Service's in-cluster FQDN or URL:
self.service.out.fqdn
self.service.out.url { scheme = "http"; port = 8080; }

# Reference a workload's pod selector while tracking the dependency:
self.deployment.out.selector

# Read a workload's pod selector without creating a dependency cycle:
self.deployment.rawSelector

# Build a workload with volume mounts using the pipe operator:
{ name = "web"; spec = { ... }; }
  |> kix.withMounts [ self.dataMount ]
  |> scope.mkDeployment

# Derive a Service from a workload's selector and container ports:
self.deployment
  |> kix.service.fromWorkload { name = "web"; }
  |> scope.mkResource

# Expose a Service via Ingress or Gateway API HTTPRoute:
(kix.expose { service = "service"; })

# Create a ServiceMonitor and optional alert rules when Prometheus is present:
(kix.monitor { service = "service"; port = "metrics"; })

# Run a post-deploy health check Job against service outputs:
(kix.healthCheck {
  script = builtins.readFile ./health-check.py;
  env = { self, ... }: { TARGET_URL = self.service.out.url { }; };
})

# Reference a key from a Kix-managed Secret:
self.credentials.out.keyRef "password"

# Reference a key from an externally managed Secret:
(scope.mkSecretRef {
  name = "external-db-creds";
  keys = [ "username" "password" ];
}).out.keyRef "password"

# Read a container image from a committed pin file and render repo:tag@digest:
# (in cluster config) config.image = pins.get "web";
# (in package build) image = kix.image.ref config.image;

# Convert an attribute set of environment variables to a Kubernetes env list, dropping nulls:
kix.mkEnvVars {
  LOG_LEVEL = config.logLevel;
  OPTIONAL_FLAG = config.optionalFlag or null;
  DB_PASSWORD = database.out.credentials.out.keyRef "password";
}

# Gate an optional resource on a boolean condition:
scope.mkOptional config.metrics.enabled (_: scope.mkResource { ... })

# Write an intentional empty map (such as emptyDir) that must not be stripped:
emptyDir = { };

# Prevent a constructor-built map from collapsing when all optional fields are null:
kix.keep {
  medium = config.medium;
  sizeLimit = config.sizeLimit;
}
```
